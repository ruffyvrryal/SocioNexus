Run this in your Supabase SQL Editor to prepare tables:

CREATE TABLE accounts (
  id TEXT PRIMARY KEY,
  name TEXT,
  handle TEXT,
  category TEXT,
  avatar_url TEXT
);

CREATE TABLE content_posts (
  id TEXT PRIMARY KEY,
  account_id TEXT,
  platform TEXT,
  post_date DATE,
  caption TEXT,
  impressions INT DEFAULT 0,
  reach INT DEFAULT 0,
  likes INT DEFAULT 0,
  comments INT DEFAULT 0,
  shares INT DEFAULT 0,
  saves INT DEFAULT 0
);
