Supabase Cloud Database

Data is synced across all devices using Supabase. Check your configuration below or update keys.

Click to view required SQL setup query

Run this query in your Supabase SQL Editor to construct the aligned schema:

CREATE TABLE IF NOT EXISTS public.accounts (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  handle TEXT NOT NULL,
  category TEXT,
  avatar_url TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.content_posts (
  id TEXT PRIMARY KEY,
  account_id TEXT NOT NULL REFERENCES public.accounts(id) ON DELETE CASCADE,
  platform TEXT NOT NULL,
  post_date DATE NOT NULL,
  caption TEXT NOT NULL,
  impressions BIGINT DEFAULT 0,
  reach BIGINT DEFAULT 0,
  likes BIGINT DEFAULT 0,
  comments BIGINT DEFAULT 0,
  shares BIGINT DEFAULT 0,
  saves BIGINT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.accounts ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.content_posts ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public Access Accounts" ON public.accounts;
DROP POLICY IF EXISTS "Public Access Posts" ON public.content_posts;

CREATE POLICY "Public Access Accounts" ON public.accounts FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Access Posts" ON public.content_posts FOR ALL USING (true) WITH CHECK (true);
